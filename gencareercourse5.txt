<script>
const API_BASE = '';

let questions = [];
let current = 0;
let answers = [];
let currentData = null; // 保存匹配结果用于切换

async function loadQuestions(){
  try{
    const r = await fetch('/api/questions');
    const d = await r.json();
    questions = d.questions || [];
  }catch(e){
    alert('无法加载题目，请刷新重试');
  }
}

function startQuiz(){
  answers = [];
  current = 0;
  document.getElementById('start-screen').classList.remove('active');
  document.getElementById('quiz-screen').classList.add('active');
  loadQuestions().then(()=>showQuestion());
}

function showQuestion(){
  if(current >= questions.length){
    submitAnswers();
    return;
  }
  const q = questions[current];
  document.getElementById('q-num').textContent = current+1;
  document.getElementById('q-dim').textContent = q.dimension || '维度';
  document.getElementById('q-text').textContent = q.question;
  document.getElementById('progress-fill').style.width = ((current+1)/questions.length*100)+'%';

  const area = document.getElementById('options-area');
  area.innerHTML = '';
  (q.options||[]).forEach((opt, idx)=>{
    const btn = document.createElement('button');
    btn.className = 'option-btn';
    btn.textContent = opt.text;
    btn.onclick = ()=>{
      answers.push({question_id: q.id, option_index: idx});
      current++;
      setTimeout(showQuestion, 280);
    };
    area.appendChild(btn);
  });
}

async function submitAnswers(){
  document.getElementById('quiz-screen').classList.remove('active');
  document.getElementById('loading-screen').classList.add('active');
  try{
    const r = await fetch('/api/match', {
      method:'POST',
      headers:{'Content-Type':'application/json'},
      body:JSON.stringify({answers, top_n:10})
    });
    const data = await r.json();
    currentData = data;
    renderResults(data);
    document.getElementById('loading-screen').classList.remove('active');
    document.getElementById('result-screen').classList.add('active');
    window.scrollTo({top:0,behavior:'smooth'});
  }catch(e){
    alert('匹配请求失败，返回开始页');
    resetAll();
  }
}

/* 渲染左侧列表 + 默认选中第一个 */
function renderResults(data){
  const matches = (data.matches || []).slice(0,10);
  const left = document.getElementById('left-list');
  left.innerHTML = '';

  matches.forEach((m, idx)=>{
    const fig = m.figure || {};
    const s = m.suggestion || {};
    const item = document.createElement('div');
    item.className = 'figure-item' + (idx===0 ? ' active' : '');
    item.innerHTML = `
      <div class="figure-thumb">${(fig.name_cn||fig.name||'?').charAt(0)}</div>
      <div class="figure-info">
        <div class="name">${s.figure_name || fig.name_cn || '未知人物'}</div>
        <div class="meta">${fig.name||''} · ${fig.era||''} · ${fig.type||''}</div>
      </div>
      <div class="sim">${(m.similarity*100).toFixed(0)}%</div>
      <div class="item-clear"></div>
    `;
    item.onclick = ()=>{
      document.querySelectorAll('.figure-item').forEach(x=>x.classList.remove('active'));
      item.classList.add('active');
      showRightDetail(m, idx);
    };
    left.appendChild(item);
  });

  // 默认显示第一个
  if(matches.length > 0){
    showRightDetail(matches[0], 0);
  }
}

function showRightDetail(m, idx){
  const fig = m.figure || {};
  const s = m.suggestion || {};
  // 右侧大图视觉区
  document.getElementById('hero-avatar').textContent = (fig.name_cn||fig.name||'?').charAt(0);
  document.getElementById('hero-name').textContent = s.figure_name || fig.name_cn || '人物';
  document.getElementById('hero-sub').textContent = `${fig.name||''} · ${fig.era||''} · ${fig.type||''} · 相似度 ${(m.similarity*100).toFixed(0)}%`;

  // 文字内容
  document.getElementById('r-early').textContent = s.early_career || fig.early_career || '—';
  document.getElementById('r-actions').textContent = s.early_actions || fig.early_actions || '—';
  document.getElementById('r-break').textContent = s.breakthrough || fig.breakthrough || '—';
  document.getElementById('r-lesson').textContent = s.key_lesson || fig.key_lesson || '—';
  document.getElementById('r-overall').textContent = s.overall || '—';

  // 差距表
  const tbody = document.querySelector('#r-gaps tbody');
  tbody.innerHTML = '';
  (s.gaps||[]).forEach(g=>{
    const tr = document.createElement('tr');
    tr.innerHTML = `
      <td>${g.dimension}</td>
      <td>${g.user_value}</td>
      <td>${g.figure_value}</td>
      <td style="color:#ff6b6b;font-weight:700;">${g.direction}</td>
    `;
    tbody.appendChild(tr);
  });
}

  document.getElementById('loading-screen').classList.remove('active');
  document.getElementById('result-screen').classList.remove('active');

}

### 你需要做的 3 步修正

1. **替换文件**：把上面代码保存为 `static/index.html`（覆盖旧版）。
2. **确保后端正常**：`python -m uvicorn src.main:app --host 0.0.0.0 --port 8005`，并确认 `/api/match` 返回的 `matches` 里有 `suggestion` 字段（我已在 `matcher.py` 里写好了 `generate_suggestion`，直接用）。
3. **图片位置**：右侧 `hero-visual` 已做成**大图 + 头像 + 名字**明显显示。你只要在 `figures.json` 里给每个人加 `"avatar":"https://..."`，然后把 `hero-visual` 里的 `<img>` 替换成 `<img src="${fig.avatar||''}" ...>` 即可；现在先用 CSS 渐变 + 首字做占位，**图明确在右侧且明显**。

如果你想要左侧只显示 **Top5** 且固定高度、右侧高度与之完全一致并随内容滚动，当前代码已用 `align-items:stretch` + `position:sticky` 实现；多出的人物在左侧自然出现滚动条。
