<script>
const API_BASE = '';

let questions = [];
let current = 0;
let answers = [];
let matchData = null;

async function loadQuestions(){
  try{
    const r = await fetch('/api/questions');
    const d = await r.json();
    questions = d.questions || [];
  }catch(e){ alert('无法加载题目'); }
}

function startQuiz(){
  answers = []; current = 0;
  document.getElementById('start-screen').classList.remove('active');
  document.getElementById('quiz-screen').classList.add('active');
  loadQuestions().then(()=>showQuestion());
}

function showQuestion(){
  if(current >= questions.length){ submitAnswers(); return; }
  const q = questions[current];
  document.getElementById('q-num').textContent = current+1;
  document.getElementById('q-dim').textContent = q.dimension || '维度';
  document.getElementById('q-text').textContent = q.question;
  document.getElementById('progress-fill').style.width = ((current+1)/questions.length*100)+'%';

  const area = document.getElementById('options-area');
  area.innerHTML = '';
  (q.options||[]).forEach((opt, idx)=>{
    const btn = document.createElement('button');
    btn.className = 'option-btn';
    btn.textContent = opt.text;
    btn.onclick = ()=>{
      answers.push({question_id: q.id, option_index: idx});
      current++;
      setTimeout(showQuestion, 250);
    };
    area.appendChild(btn);
  });
}

async function submitAnswers(){
  document.getElementById('quiz-screen').classList.remove('active');
  document.getElementById('loading-screen').classList.add('active');
  try{
    const r = await fetch('/api/match', {
      method:'POST',
      headers:{'Content-Type':'application/json'},
      body:JSON.stringify({answers, top_n:10})
    });
    const data = await r.json();
    matchData = data;
    renderResultPage(data);
    document.getElementById('loading-screen').classList.remove('active');
    document.getElementById('result-screen').classList.add('active');
    window.scrollTo({top:0,behavior:'smooth'});
  }catch(e){
    alert('匹配失败，返回开始页'); resetAll();
  }
}

/* 渲染列表 + 默认选中第一个 */
function renderResultPage(data){
  const list = document.getElementById('selector-list');
  list.innerHTML = '';
  const matches = (data.matches || []).slice(0,10);

  matches.forEach((m, idx)=>{
    const fig = m.figure || {};
    const s = m.suggestion || {};
    const item = document.createElement('div');
    item.className = 'selector-item' + (idx===0 ? ' active' : '');
    item.innerHTML = `
      <div class="selector-thumb">${(fig.name_cn||fig.name||'?').charAt(0)}</div>
      <div class="selector-info">
        <div class="name">${s.figure_name || fig.name_cn || '未知'}</div>
        <div class="meta">${fig.name||''} · ${fig.era||''} · ${fig.type||''}</div>
      </div>
      <div class="selector-sim">${(m.similarity*100).toFixed(0)}%</div>
    `;
    item.onclick = ()=>{
      document.querySelectorAll('.selector-item').forEach(x=>x.classList.remove('active'));
      item.classList.add('active');
      updateHeroAndDetails(m);
    };
    list.appendChild(item);
  });

  if(matches.length > 0){
    updateHeroAndDetails(matches[0]);
  }
}

function updateHeroAndDetails(m){
  const fig = m.figure || {};
  const s = m.suggestion || {};
  // 顶部大图/视觉
  document.getElementById('hero-avatar').textContent = (fig.name_cn||fig.name||'?').charAt(0);
  document.getElementById('hero-name').textContent = s.figure_name || fig.name_cn || '人物';
  document.getElementById('hero-sub').textContent = `${fig.name||''} · ${fig.era||''} · ${fig.type||''}`;
  document.getElementById('hero-sim').textContent = (m.similarity*100).toFixed(0)+'%';

  // 底部详情
  document.getElementById('d-early').textContent = s.early_career || fig.early_career || '—';
  document.getElementById('d-actions').textContent = s.early_actions || fig.early_actions || '—';
  document.getElementById('d-break').textContent = s.breakthrough || fig.breakthrough || '—';
  document.getElementById('d-lesson').textContent = s.key_lesson || fig.key_lesson || '—';
  document.getElementById('d-overall').textContent = s.overall || '—';

  // 差距表
  const tbody = document.querySelector('#d-gaps tbody');
  tbody.innerHTML = '';
  (s.gaps||[]).forEach(g=>{
    const tr = document.createElement('tr');
    tr.innerHTML = `<td>${g.dimension||''}</td><td>${g.user_value}</td><td>${g.figure_value}</td><td style="color:#ff6b6b;font-weight:700;">${g.direction||''}</td>`;
    tbody.appendChild(tr);
  });
}

function resetAll(){
  current = 0; answers = []; matchData = null;
  document.getElementById('start-screen').classList.add('active');
  document.getElementById('quiz-screen').classList.remove('active');
  document.getElementById('loading-screen').classList.remove('active');
  document.getElementById('result-screen').classList.remove('active');
  document.getElementById('selector-list').innerHTML = '';
}
</script>

**关键改动说明**：
- `result-screen` 现在是 **单列**：`hero-visual`（图在上）→ `selector-list`（点击列表在中）→ `detail-block`（详情在下）
- 点击 `selector-item` 只调用 `updateHeroAndDetails()`，同步刷新上方大图和下方内容，无需重新渲染整个页面
- 手机端 `max-width:720px`，图片、字、列表都紧凑适配，滚动自然

把文件覆盖后刷新 `http://localhost:8005` 测试即可。若需要把“图”换成真实图片URL，直接在 `match.py` 返回的 `figure` 里加 `"avatar":"https://..."`，然后把 `updateHeroAndDetails` 里 `hero-avatar` 改为 `<img src="${fig.avatar||''}">` 即可。
