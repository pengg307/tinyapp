### 5. `src/engine/matcher.py`

```python
import json
import math
from pathlib import Path

DIMENSIONS = [
    "openness","conscientiousness","extraversion","agreeableness","neuroticism",
    "leadership","risk_taking","rationality","discipline","empathy","ambition","resilience"
]

DATA_DIR = Path("src/data")

def load_figures():
    with open(DATA_DIR/"figures.json","r",encoding="utf-8") as f:
        return json.load(f)["figures"]

def load_questions():
    with open(DATA_DIR/"questions.json","r",encoding="utf-8") as f:
        return json.load(f)["questions"]

def calculate_user_vector(answers):
    # answers: [{"question_id":1,"option_index":0}, ...]
    dims = {d:0.0 for d in DIMENSIONS}
    counts = {d:0 for d in DIMENSIONS}
    questions = load_questions()
    q_map = {q["id"]:q for q in questions}
    for ans in answers:
        qid = ans.get("question_id")
        idx = max(0, min(len(q_map[qid]["options"])-1, ans.get("option_index",0)))
        opt = q_map[qid]["options"][idx]
        for d,val in opt.get("values",{}).items():
            if d in dims:
                dims[d] += val
                counts[d] += 1
    # 平均
    for d in dims:
        dims[d] = dims[d]/max(counts[d],1)
    return dims

def euclidean_distance(v1, v2):
    s = 0.0
    for d in DIMENSIONS:
        s += (v1.get(d,0)-v2.get(d,0))**2
    return math.sqrt(s)

def is_real_figure(fig):
    # 真实人物ID通常不含 figure_ 前缀，或可根据预设列表判断
    real_ids = {"newton","einstein","tesla","galileo","mozart","shakespeare","curie","lincoln","steve_jobs","elon_musk","mark_zuckerberg","bill_gates","warren_buffett","edison","freud","darwin","pasteur","ada_lovelace","florence_nightingale","helen_keller"}
    return fig.get("id") in real_ids

def match_user(answers, top_n=10):
    user_vec = calculate_user_vector(answers)
    figures = load_figures()
    results = []
    for fig in figures:
        dist = euclidean_distance(user_vec, fig["vector"])
        # 相似度公式
        sim = 1.0 - min(dist / math.sqrt(len(DIMENSIONS)), 1.0)
        if is_real_figure(fig):
            sim += 0.20
        results.append({
            "figure": fig,
            "similarity": round(sim, 3),
            "distance": round(dist, 3)
        })
    results.sort(key=lambda x: -x["similarity"])
    top = results[:top_n]
    # 生成建议
    for r in top:
        r["suggestion"] = generate_suggestion(r, user_vec)
    return {"user_vector": user_vec, "matches": top}

def generate_suggestion(match_result, user_vec):
    fig = match_result["figure"]
    gaps = []
    for d in DIMENSIONS:
        u = user_vec.get(d,0)
        f = fig["vector"].get(d,0)
        diff = round(abs(u-f),2)
        if diff > 0.15:
            direction = "develop" if f > u else "leverage"
            gaps.append({
                "dimension": d,
                "user_value": round(u,2),
                "figure_value": round(f,2),
                "difference": diff,
                "direction": direction
            })
    gaps.sort(key=lambda x: -x["difference"])
    overall = f"Your career profile is most similar to {fig['name_cn']} ({fig['name']}, {fig['era']})."
    return {
        "figure_name": fig["name_cn"],
        "figure_name_en": fig["name"],
        "era": fig["era"],
        "type": fig["type"],
        "early_career": fig["early_career"],
        "early_actions": fig["early_actions"],
        "breakthrough": fig["breakthrough"],
        "key_lesson": fig["key_lesson"],
        "gaps": gaps[:5],
        "overall": overall
    }
```

### 8. `src/api/match.py`

```python
from fastapi import APIRouter
from pydantic import BaseModel
from typing import List, Optional
from src.engine.matcher import match_user

router = APIRouter()

class AnswerItem(BaseModel):
    question_id: int
    option_index: int

class MatchRequest(BaseModel):
    answers: List[AnswerItem]
    top_n: Optional[int] = 10

@router.post("/match")
async def post_match(req: MatchRequest):
    answers = [{"question_id": a.question_id, "option_index": a.option_index} for a in req.answers]
    result = match_user(answers, top_n=req.top_n or 10)
    return result
```

const API_BASE = ''; // 同源

let questions = [];
let current = 0;
let answers = [];

async function loadQuestions() {
  try {
    const r = await fetch('/api/questions');
    const d = await r.json();
    questions = d.questions || [];
  } catch(e){
    alert('无法加载题目，请刷新页面重试');

function startQuiz(){
  answers = [];
  current = 0;
  document.getElementById('start-screen').classList.remove('active');
  document.getElementById('quiz-screen').classList.add('active');
  loadQuestions().then(() => showQuestion());

function showQuestion(){
  if(current >= questions.length){
    submitAnswers();
    return;
  }
  const q = questions[current];
  document.getElementById('q-num').textContent = current+1;
  document.getElementById('q-dim').textContent = q.dimension || '性格维度';
  document.getElementById('q-text').textContent = q.question;
  const pct = ((current+1)/questions.length)*100;
  document.getElementById('progress-fill').style.width = pct+'%';

  const area = document.getElementById('options-area');
  area.innerHTML = '';

    btn.className = 'option-btn';
    btn.textContent = opt.text;
    btn.onclick = () => {

      setTimeout(showQuestion, 300);

    area.appendChild(btn);
  });
}

          <strong>特质差距（建议发展方向）：</strong>
          ${(s.gaps||[]).map(g=>`
            <div class="gap-row"><span class="gap-label">${g.dimension}</span><span>你 ${g.user_value} → 他 ${g.figure_value} (${g.direction})</span></div>
          `).join('')}
        </div>
        <p style="margin-top:.6rem;color:#ff6b6b;font-weight:700;">${s.overall || ''}</p>
      </div>
    `;
    card.onclick = () => {
      const d = card.querySelector('.details');
      d.style.display = d.style.display === 'none' ? 'block' : 'none';
    };
    list.appendChild(card);
  });
}
